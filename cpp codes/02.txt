class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        vector<int> ans;

        int product = 1;
        for (auto num : nums) {
            ans.push_back(product);
            product *= num;
        }

        product = 1;
        for (int i = nums.size() - 1; i >= 0; i--) {
            ans[i] = product * ans[i];
            product *= nums[i];
        }

        return ans;
    }
};
