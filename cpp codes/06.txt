class Solution {
public:
    vector<string> findWords(vector<string>& words) {
        unordered_set<char> row1 = {'q', 'w', 'e', 'r', 't',
                                    'y', 'u', 'i', 'o', 'p'};
        unordered_set<char> row2 = {'a', 's', 'd', 'f', 'g',
                                    'h', 'j', 'k', 'l'};
        unordered_set<char> row3 = {'z', 'x', 'c', 'v', 'b', 'n', 'm'};

        vector<string> ans;

        for (auto word : words) {
            int row = 0;
            bool flag = 1;

            for (auto ch : word) {
                ch = tolower(ch);

                if (row1.find(ch) != row1.end()) {
                    if ((row != 0 && row != 1)) {
                        flag = 0;
                        break;
                    }
                    row = 1;
                } else if (row2.find(ch) != row2.end()) {
                    if ((row != 0 && row != 2)) {
                        flag = 0;
                        break;
                    }
                    row = 2;
                } else if (row3.find(ch) != row3.end()) {
                    if ((row != 0 && row != 3)) {
                        flag = 0;
                        break;
                    }
                    row = 3;
                }
            }
            
            if (flag == 1) {
                ans.push_back(word);
            }
        }

        return ans;
    }
};
