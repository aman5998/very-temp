https://leetcode.com/problems/sort-array-by-parity/

class Solution {
public:
    vector<int> sortArrayByParity(vector<int>& nums) {
        int i = 0;
        int j = 0;

        while (i < nums.size() && j < nums.size()) {
            if (nums[j] & 1) {
                j++;
            } else {
                if (nums[i] & 1) {
                    swap(nums[i], nums[j]);
                    i++, j++;
                } else {
                    i++;
                }
            }
        }

        return nums;
    }
};
